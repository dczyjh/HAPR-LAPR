def gate(method):

    assert torch.cuda.is_available() and torch.cuda.device_count()==1

    teacher={n:q.tensor_hash(v) for n,v in tr.model_teacher.state_dict().items()}

    batch=next(iter(tr.train_loader_x));identity={k:q.tensor_hash(batch[k]) for k in ('img','label')};identity['impath']=batch['impath']

    for e,b in ((0,0),(1,0),(1,351),(2,0)):

    for added in (False,True) if method=='o4' else (False,):assert any(v>0 for n,v in changed.items() if q.added(n)==added)
    before={n:q.tensor_hash(v) for n,v in tr.model.state_dict().items()};tr.save_model(2,str(out),val_result=0.,model_name='model-best.pth.tar');tr.load_model(str(out))
