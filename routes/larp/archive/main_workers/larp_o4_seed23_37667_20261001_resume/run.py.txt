def compare_first(a,b,label):
    if label!='student_logits':return q.compare_tensor(a,b,'forward',label)
    assert a.shape==b.shape and a.dtype==b.dtype and a.ndim==2,label
    assert torch.isfinite(a).all() and torch.isfinite(b).all(),label
    x,y=a.float(),b.float();delta=(x-y).abs()
    bound=.002+.002*torch.maximum(x.abs().amax(-1,keepdim=True),y.abs().amax(-1,keepdim=True))
    bad=delta>bound;old=delta>.002+.002*y.abs()
    return dict(label=label,kind='forward',passed=not bool(bad.any()),max_abs=float(delta.max()),failed_elements=int(bad.sum()),
        elementwise_failed_diagnostic=int(old.sum()),prediction_changes=int((x.argmax(-1)!=y.argmax(-1)).sum()),criterion='symmetric_row_scaled_original_reload_bound')
def repair_tests():
    data=[torch.load(OLD/'gate/seed_2/fgvc_aircraft'/r/'first_step.pt',map_location='cpu') for r in ('o4','r1')]
    checks={k:compare_first(data[0][k],data[1][k],k) for k in data[0]}
    assert all(v['passed'] for v in checks.values())
    assert checks['student_logits']['elementwise_failed_diagnostic']==3 and checks['student_logits']['prediction_changes']==0
    a=torch.tensor([[4.,1.]],dtype=torch.float32);b=a.clone()
    assert compare_first(a,b,'student_logits')['passed']
    b[0,1]+=.02;assert not compare_first(a,b,'student_logits')['passed']
    assert compare_first(a,b,'student_logits')['failed_elements']==compare_first(b,a,'student_logits')['failed_elements']
    rejected=0
    for other in (a.double(),a[:,:1],torch.tensor([[float('nan'),1.]]),torch.tensor([[float('inf'),1.]])):
        try:compare_first(a,other,'student_logits')
        except AssertionError:rejected+=1
    assert rejected==4
    old=ast.parse((OLD/'run.py').read_text());new=ast.parse((ROOT/'run.py').read_text().replace('1.79','1.78'))
    def node(tree,name):return next(z for z in tree.body if isinstance(z,ast.FunctionDef) and z.name==name)
    same=('build','config','train_run','evaluate','reserve','ident')
    for name in same:assert ast.dump(node(old,name))==ast.dump(node(new,name)),name
    gate=node(new,'gate')
    class RestoreCall(ast.NodeTransformer):
        def visit_Call(self,z):
            self.generic_visit(z)
            if isinstance(z.func,ast.Name) and z.func.id=='compare_first':
                z.func=ast.Attribute(value=ast.Name(id='q',ctx=ast.Load()),attr='compare_tensor',ctx=ast.Load())
                z.args.insert(2,ast.Constant(value='forward'))
            return z
    assert ast.dump(node(old,'gate'))==ast.dump(RestoreCall().visit(gate))
    return dict(status='passed',saved_failure_checks=checks,synthetic_rejections=rejected,over_bound_rejected=True,symmetric=True,scientific_functions_exact=list(same),gate_only_comparison_changed=True)
