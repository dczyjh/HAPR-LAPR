def normalize(v):
    if isinstance(v,str) and str(B)!=str(HOST): return v.replace(str(B)+'/',str(HOST)+'/')
    if isinstance(v,list):return [normalize(x) for x in v]
    if isinstance(v,dict):return {k:normalize(x) for k,x in v.items()}
    return v
def relocate(v):
    if isinstance(v,str) and v.startswith(str(HOST)+'/'):return str(B)+v[len(str(HOST)):]
    if isinstance(v,list):return [relocate(x) for x in v]
    if isinstance(v,dict):return {k:relocate(x) for k,x in v.items()}
    return v

class Wiring(ast.NodeTransformer):
    def visit_Subscript(self,node):
        self.generic_visit(node)
        if ast.unparse(node)=="q.read(NODE / 'cpu_verified.json')['lr_trajectories']":return ast.parse("q.read(ROOT/'preflight.json')['lr_trajectories']",mode='eval').body
        return node
    def visit_Assign(self,node):
        self.generic_visit(node)
        if any(isinstance(x,ast.Name) and x.id=='ref' for x in node.targets) and ast.unparse(node.value)=="ROOT / 'formal' / f'seed_{SEED}' / ds / 'r1'":node.value=ast.parse('r1_path(ds)',mode='eval').body
        return node
    def visit_Compare(self,node):
        self.generic_visit(node)
        if ast.unparse(node)=="identity == pair[i]['identity']":return ast.parse("identity == normalize(pair[i]['identity'])",mode='eval').body
        return node
    def visit_BinOp(self,node):
        self.generic_visit(node)
        if ast.unparse(node)=='(b @ a).norm() / math.sqrt(2)':return ast.parse('(b @ a).norm() * mod.scaling',mode='eval').body
        return node

def operator_tests():
    from trainers.efficient_adaptation import WeightLoRAParametrization
    checks=[]
    for rank in (1,2,4):
        torch.manual_seed(17);base=torch.randn(16,16)
        op=WeightLoRAParametrization(16,16,rank,1.,torch.float32,torch.device('cpu'))
        assert op.scaling==1/math.sqrt(rank) and sum(p.numel() for p in op.parameters())==32*rank
        assert torch.equal(op(base),base)
        opt=torch.optim.SGD(op.parameters(),lr=.01,momentum=.9,weight_decay=.0005)
        for _ in range(4):opt.zero_grad();op(base).square().sum().backward();opt.step()
        assert all(p.grad is not None and torch.isfinite(p.grad).all() and torch.count_nonzero(p.grad)>0 for p in op.parameters())
        assert torch.count_nonzero(op.lora_B_o@op.lora_A_o)>0
        clone=copy.deepcopy(op);clone.load_state_dict(op.state_dict());assert torch.equal(op(base),clone(base))
        checks.append(rank)
    a=torch.tensor([[4.,1.]]);b=a.clone();b[0,1]+=.02
    assert compare_first(a,a,'student_logits')['passed'] and not compare_first(a,b,'student_logits')['passed']
    return dict(status='passed',ranks=checks,zero_B_exact=True,update_and_roundtrip=True,overbound_rejected=True)
